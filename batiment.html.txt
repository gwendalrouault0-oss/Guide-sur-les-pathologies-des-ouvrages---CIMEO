  <script>
    const buildingWebcam = document.getElementById('buildingWebcam');
    const buildingCanvas = document.getElementById('buildingSnapshot');
    const buildingCtx = buildingCanvas.getContext('2d');
    const buildingImage = document.getElementById('buildingImage');
    const buildingForm = document.getElementById('batimentForm');
    const buildingResult = document.getElementById('buildingResult');
    const buildingCameraStatus = document.getElementById('buildingCameraStatus');
    const buildingCameraLabel = document.getElementById('buildingCameraLabel');
    
    let buildingStream = null;
    let capturedImageData = null;
    let isMobile = /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(navigator.userAgent);
    let photoTaken = false;
    let isAnalyzing = false;
    let isBackCamera = true; // true = arrière, false = avant

    // Initialiser la webcam
    async function setupBuildingWebcam() {
      try {
        if (!navigator.mediaDevices?.getUserMedia) return;

        if (buildingStream) {
          buildingStream.getTracks().forEach(track => track.stop());
        }

        const constraints = {
          video: {
            width: { ideal: isMobile ? 1280 : 1920 },
            height: { ideal: isMobile ? 720 : 1080 },
            facingMode: isBackCamera ? 'environment' : 'user'
          }
        };

        buildingStream = await navigator.mediaDevices.getUserMedia(constraints);
        buildingWebcam.srcObject = buildingStream;
        await buildingWebcam.play();

        // Mettre à jour le statut de la caméra
        buildingCameraStatus.style.display = 'flex';
        buildingCameraLabel.textContent = isBackCamera ? '📹 Caméra arrière' : '📹 Caméra frontale';
        
        // Activer le bouton de changement de caméra
        document.getElementById('switchCameraBuilding').disabled = false;

      } catch (error) {
        console.warn('Accès webcam non disponible', error);
        // Fallback
        try {
          const fallbackConstraints = {
            video: { 
              width: { ideal: 640 },
              height: { ideal: 480 }
            }
          };
          buildingStream = await navigator.mediaDevices.getUserMedia(fallbackConstraints);
          buildingWebcam.srcObject = buildingStream;
          await buildingWebcam.play();
          
          buildingCameraStatus.style.display = 'flex';
          buildingCameraLabel.textContent = '📹 Caméra par défaut';
          document.getElementById('switchCameraBuilding').disabled = false;
        } catch (fallbackError) {
          console.warn('Fallback camera failed', fallbackError);
          document.getElementById('switchCameraBuilding').disabled = true;
        }
      }
    }

    // Changer de caméra
    document.getElementById('switchCameraBuilding').addEventListener('click', async (e) => {
      e.preventDefault();
      if (photoTaken) return;
      
      // Basculer entre arrière et avant
      isBackCamera = !isBackCamera;
      console.log('Changement vers caméra:', isBackCamera ? 'arrière' : 'avant');
      
      // Réinitialiser la webcam avec la nouvelle caméra
      await setupBuildingWebcam();
    });

    // Capturer depuis webcam
    document.getElementById('captureBuilding').addEventListener('click', (e) => {
      e.preventDefault();
      
      if (photoTaken && !isAnalyzing) {
        // Si une photo a déjà été prise et qu'on n'est pas en train d'analyser, on analysea
        analyzeBuilding();
        return;
      }
      
      if (!buildingStream) {
        alert('Caméra non disponible');
        return;
      }
      
      // Prendre la photo
      buildingCanvas.width = buildingWebcam.videoWidth;
      buildingCanvas.height = buildingWebcam.videoHeight;
      buildingCtx.drawImage(buildingWebcam, 0, 0);
      const imageData = buildingCanvas.toDataURL('image/jpeg', 0.95);
      buildingImage.src = imageData;
      capturedImageData = imageData;
      
      // Afficher l'image et masquer la webcam
      buildingImage.style.display = 'block';
      buildingImage.classList.add('photo-taken');
      buildingWebcam.style.display = 'none';
      buildingCameraStatus.style.display = 'none';
      
      // Marquer qu'une photo a été prise
      photoTaken = true;
      
      // Changer le texte du bouton
      document.getElementById('captureBuilding').textContent = '🔍 Analyser avec cette photo';
      
      // Désactiver le bouton de changement de caméra
      document.getElementById('switchCameraBuilding').disabled = true;
    });

    // Fonction d'analyse du bâtiment
    function analyzeBuilding() {
      if (!capturedImageData) return;
      
      isAnalyzing = true;
      
      // Simuler l'analyse (remplacer par vraie logique d'analyse)
      setTimeout(() => {
        // L'analyse garde automatiquement la photo
        isAnalyzing = false;
        
        // Ici vous pouvez ajouter la logique d'analyse réelle
        alert('Analyse terminée ! La photo a été conservée pour le diagnostic.');
      }, 2000);
    }

    // Upload fichier
    document.getElementById('uploadBuildingBtn').addEventListener('click', () => {
      document.getElementById('buildingFileInput').click();
    });

    document.getElementById('buildingUpload').addEventListener('click', () => {
      document.getElementById('buildingFileInput').click();
    });

    document.getElementById('buildingFileInput').addEventListener('change', (e) => {
      const file = e.target.files[0];
      if (!file || !file.type.startsWith('image/')) {
        alert('Veuillez sélectionner une image valide');
        return;
      }
      const reader = new FileReader();
      reader.onload = (ev) => {
        buildingImage.src = ev.target.result;
        capturedImageData = ev.target.result;
        buildingImage.style.display = 'block';
        buildingImage.classList.add('photo-taken');
        buildingWebcam.style.display = 'none';
        buildingCameraStatus.style.display = 'none';
        
        photoTaken = true;
        document.getElementById('captureBuilding').textContent = '🔍 Analyser avec cette photo';
        document.getElementById('switchCameraBuilding').disabled = true;
      };
      reader.readAsDataURL(file);
    });

    // Drag & Drop
    document.getElementById('buildingUpload').addEventListener('dragover', (e) => {
      e.preventDefault();
      e.target.classList.add('dragover');
    });

    document.getElementById('buildingUpload').addEventListener('dragleave', (e) => {
