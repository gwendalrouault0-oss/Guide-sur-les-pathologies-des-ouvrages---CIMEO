  <script>
    // Utilisateurs de démonstration
    const demoUsers = {
      'client@demo.com': {
        password: 'demo123',
        type: 'client',
        name: 'Client Démo',
        company: 'Entreprise Test'
      },
      'employe@colza.fr': {
        password: 'demo123',
        type: 'employee',
        name: 'Employé COLZA',
        company: 'COLZA Normandie'
      },
      'admin@colza.fr': {
        password: 'demo123',
        type: 'admin',
        name: 'Administrateur',
        company: 'COLZA Normandie'
      }
    };

    // Gestion de la connexion
    document.getElementById('loginForm').addEventListener('submit', function(e) {
      e.preventDefault();
      
      const email = document.getElementById('email').value;
      const password = document.getElementById('password').value;
      const userType = document.getElementById('userType').value;
      
      // Vérification des credentials
      if (demoUsers[email] && 
          demoUsers[email].password === password && 
          demoUsers[email].type === userType) {
        
        // Sauvegarder la session
        const userData = {
          email: email,
          name: demoUsers[email].name,
          type: userType,
          company: demoUsers[email].company,
          loginTime: new Date().toISOString()
        };
        
        sessionStorage.setItem('currentUser', JSON.stringify(userData));
        
        showMessage('Connexion réussie ! Redirection...', 'success');
        
        // Redirection selon le type d'utilisateur
        setTimeout(() => {
          switch(userType) {
            case 'client':
              window.location.href = 'dashboard-client.html';
              break;
            case 'employee':
              window.location.href = 'dashboard-employee.html';
              break;
            case 'admin':
              window.location.href = 'dashboard-admin.html';
              break;
            default:
              window.location.href = 'index.html';
          }
        }, 1500);
        
      } else {
        showMessage('Email, mot de passe ou type de compte incorrect', 'error');
      }
    });

    // Affichage des messages
    function showMessage(message, type) {
      const messageDiv = document.getElementById('loginMessage');
      messageDiv.textContent = message;
      messageDiv.className = `login-message ${type}`;
      messageDiv.style.display = 'block';
      
      if (type === 'error') {
        setTimeout(() => {
          messageDiv.style.display = 'none';
        }, 5000);
      }
    }

    // Remplir les comptes démo
    function fillDemoAccount(type) {
      const accounts = {
        'client': { email: 'client@demo.com', type: 'client' },
        'employee': { email: 'employe@colza.fr', type: 'employee' },
        'admin': { email: 'admin@colza.fr', type: 'admin' }
      };
      
      document.getElementById('email').value = accounts[type].email;
      document.getElementById('password').value = 'demo123';
      document.getElementById('userType').value = accounts[type].type;
    }

    // Gestion des modals
    function showResetPassword() {
      document.getElementById('resetModal').style.display = 'flex';
    }

    function hideResetPassword() {
      document.getElementById('resetModal').style.display = 'none';
    }

    function showRequestAccess() {
      document.getElementById('accessModal').style.display = 'flex';
    }

    function hideRequestAccess() {
      document.getElementById('accessModal').style.display = 'none';
    }

    function sendResetEmail() {
      const email = document.getElementById('resetEmail').value;
      if (email) {
        alert('Un email de réinitialisation a été envoyé à ' + email);
        hideResetPassword();
      } else {
        alert('Veuillez entrer une adresse email valide');
      }
    }

    // Fermer les modals en cliquant à l'extérieur
    window.onclick = function(event) {
      const resetModal = document.getElementById('resetModal');
      const accessModal = document.getElementById('accessModal');
      
      if (event.target === resetModal) {
        hideResetPassword();
      }
      if (event.target === accessModal) {
        hideRequestAccess();
      }
    }

    // Vérifier si déjà connecté
    window.onload = function() {
      const currentUser = sessionStorage.getItem('currentUser');
      if (currentUser) {
        const userData = JSON.parse(currentUser);
        showMessage('Vous êtes déjà connecté. Redirection...', 'success');
        setTimeout(() => {
          switch(userData.type) {
            case 'client':
              window.location.href = 'dashboard-client.html';
              break;
            case 'employee':
              window.location.href = 'dashboard-employee.html';
              break;
            case 'admin':
              window.location.href = 'dashboard-admin.html';
              break;
          }
        }, 1000);
      }
    }
  </script>
