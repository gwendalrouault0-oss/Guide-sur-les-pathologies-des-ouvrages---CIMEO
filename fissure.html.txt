  <script>
    // Variables globales pour Teachable Machine
    let model = null;
    let webcam = null;
    let canvas = null;
    let ctx = null;
    let stream = null;
    let currentFacingMode = "user"; // Caméra avant par défaut

    // Configuration du modèle et des classes
    const config = {
      modelURL: 'https://teachablemachine.withgoogle.com/models/rVulDxd6m/',
      classes: ["Structure saine", "Fissure détectée"],
      pdfs: [null, "https://gwendalrouault0-oss.github.io/Guide-sur-les-pathologies-des-ouvrages---CIMEO/pdf/Fissuration.pdf"]
    };

    // Éléments DOM
    const elements = {
      webcam: document.getElementById('fissureWebcam'),
      canvas: document.getElementById('fissureCanvas'),
      image: document.getElementById('fissureImage'),
      captureBtn: document.getElementById('captureBtn'),
      uploadBtn: document.getElementById('uploadBtn'),
      switchCameraBtn: document.getElementById('switchCameraBtn'),
      fileInput: document.getElementById('fileInput'),
      uploadArea: document.getElementById('fissureUpload'),
      modelStatus: document.getElementById('modelStatus'),
      resultSection: document.getElementById('resultSection'),
      prediction: document.getElementById('prediction'),
      confidence: document.getElementById('confidence'),
      pdfLink: document.getElementById('pdfLink'),
      pdfSection: document.getElementById('pdfSection'),
      newAnalysisBtn: document.getElementById('newAnalysisBtn'),
      resultIcon: document.getElementById('resultIcon'),
      resultTitle: document.getElementById('resultTitle'),
      resultDescription: document.getElementById('resultDescription'),
      cameraInfo: document.getElementById('fissureCameraInfo'),
      cameraLabel: document.getElementById('fissureCameraLabel'),
      cameraSelection: document.getElementById('cameraSelection'),
      cameraOptions: document.getElementById('cameraOptions')
    };

    let availableCameras = [];
    let selectedCameraId = null;
    let isMobile = /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(navigator.userAgent);

    // Initialisation du modèle IA
    async function initializeModel() {
      try {
        elements.modelStatus.className = 'model-status loading';
        elements.modelStatus.textContent = '🤖 Chargement du modèle IA...';
        
        model = await tmImage.load(
          config.modelURL + 'model.json',
          config.modelURL + 'metadata.json'
        );
        
        elements.modelStatus.className = 'model-status ready';
        elements.modelStatus.textContent = '✅ Modèle IA prêt !';
        
        await setupWebcam();
        
      } catch (error) {
        console.error('Erreur lors du chargement du modèle:', error);
        elements.modelStatus.className = 'model-status error';
        elements.modelStatus.textContent = '❌ Erreur de chargement du modèle IA';
      }
    }

    // Fonction pour détecter le type de caméra
    function getCameraType(label) {
      const lowerLabel = label.toLowerCase();
      if (lowerLabel.includes('front') || lowerLabel.includes('user') || lowerLabel.includes('facetime')) {
        return { type: 'front', icon: '🤳', name: 'Caméra frontale' };
      } else if (lowerLabel.includes('back') || lowerLabel.includes('rear') || lowerLabel.includes('environment')) {
        return { type: 'back', icon: '📱', name: 'Caméra arrière' };
      } else {
        return { type: 'unknown', icon: '📹', name: label || 'Caméra' };
      }
    }

    // Énumérer les caméras disponibles
    async function getCameras() {
      try {
        const devices = await navigator.mediaDevices.enumerateDevices();
        availableCameras = devices.filter(device => device.kind === 'videoinput');
        
        if (availableCameras.length > 1) {
          displayCameraOptions();
        } else if (availableCameras.length === 1) {
          selectedCameraId = availableCameras[0].deviceId;
        }
        
        return availableCameras;
      } catch (error) {
        console.warn('Impossible d\'énumérer les caméras', error);
        return [];
      }
    }

    // Afficher les options de caméra
    function displayCameraOptions() {
      if (availableCameras.length <= 1) return;

      elements.cameraOptions.innerHTML = '';
      
      availableCameras.forEach((camera, index) => {
        const cameraInfo = getCameraType(camera.label);
        
        const option = document.createElement('label');
        option.className = 'camera-option';
        option.setAttribute('tabindex', '0');
        
        const radio = document.createElement('input');
        radio.type = 'radio';
        radio.name = 'cameraChoice';
        radio.value = camera.deviceId;
        radio.checked = index === 0;
        
        if (index === 0) {
          selectedCameraId = camera.deviceId;
          option.classList.add('active');
        }
        
        const span = document.createElement('span');
        span.textContent = `${cameraInfo.icon} ${cameraInfo.name}`;
        
        option.appendChild(radio);
        option.appendChild(span);
        
        const selectCamera = () => {
          document.querySelectorAll('.camera-option').forEach(opt => opt.classList.remove('active'));
          option.classList.add('active');
          selectedCameraId = camera.deviceId;
          radio.checked = true;
          setupWebcam();
        };
        
        option.addEventListener('click', selectCamera);
        option.addEventListener('keydown', (e) => {
          if (e.key === 'Enter' || e.key === ' ') {
            e.preventDefault();
            selectCamera();
          }
        });
        
        elements.cameraOptions.appendChild(option);
      });
      
      elements.cameraSelection.style.display = 'block';
    }

    // Configuration de la webcam
    async function setupWebcam() {
      try {
        if (!navigator.mediaDevices?.getUserMedia) {
          throw new Error("Caméra non supportée");
        }

        if (stream) {
          stream.getTracks().forEach(track => track.stop());
        }

        const cameras = await getCameras();
        if (cameras.length === 0) return;

        const constraints = {
          video: {
            width: { ideal: isMobile ? 1280 : 1920 },
            height: { ideal: isMobile ? 720 : 1080 },
            facingMode: currentFacingMode
          }
        };

        if (selectedCameraId) {
          constraints.video.deviceId = { exact: selectedCameraId };
          delete constraints.video.facingMode;
        }

        stream = await navigator.mediaDevices.getUserMedia(constraints);
        elements.webcam.srcObject = stream;
        await elements.webcam.play();

        canvas = elements.canvas;
        ctx = canvas.getContext('2d');
        elements.captureBtn.disabled = false;

        const currentCamera = cameras.find(cam => cam.deviceId === selectedCameraId) || cameras[0];
        if (currentCamera) {
          elements.cameraInfo.style.display = 'block';
          const cameraInfo = getCameraType(currentCamera.label);
          elements.cameraLabel.textContent = `${cameraInfo.icon} ${cameraInfo.name}`;
        }

      } catch (error) {
        console.warn('Erreur d\'accès à la webcam:', error);
        elements.captureBtn.disabled = true;
        handleCameraError(error);
      }
    }

    // Gestion des erreurs de caméra
    function handleCameraError(error) {
      let message = 'Erreur caméra: ';
      
      switch(error.name) {
        case 'NotAllowedError':
          message += 'Accès refusé. Autorisez la caméra dans les paramètres.';
          break;
        case 'NotFoundError':
          message += 'Aucune caméra détectée.';
          break;
        case 'NotReadableError':
          message += 'Caméra utilisée par une autre application.';
          break;
        default:
          message += 'Utilisez "Analyser une image" à la place.';
      }
      
      elements.modelStatus.className = 'model-status error';
      elements.modelStatus.textContent = message;
    }

    // Affichage des résultats
    function displayResults(classIndex, confidence) {
      const className = config.classes[classIndex];
      const confidencePercent = (confidence * 100).toFixed(1);
      const isCrackDetected = classIndex === 1;

      elements.prediction.textContent = className;
      elements.confidence.textContent = `${confidencePercent}%`;

      // Mise à jour de l'interface selon le résultat
      if (isCrackDetected) {
        elements.resultSection.className = 'result-section crack-detected show';
        elements.resultIcon.textContent = '⚠️';
        elements.resultIcon.className = 'result-icon detection-success';
        elements.resultTitle.textContent = 'Fissure détectée';
        elements.resultDescription.textContent = 'L\'Intelligence Artificielle a identifié une fissuration sur la structure analysée.';
        
        // Afficher le lien PDF
        if (config.pdfs[1]) {
          elements.pdfLink.href = config.pdfs[1];
          elements.pdfSection.style.display = 'block';
        }
      } else {
        elements.resultSection.className = 'result-section show';
        elements.resultIcon.textContent = '✅';
        elements.resultIcon.className = 'result-icon no-detection';
        elements.resultTitle.textContent = 'Structure saine';
        elements.resultDescription.textContent = 'L\'analyse par IA n\'a pas détecté de fissuration significative.';
        
        // Masquer le lien PDF
        elements.pdfSection.style.display = 'none';
      }

      // Afficher les infos complémentaires si renseignées
      const material = document.getElementById('materialType').value;
      const location = document.getElementById('location').value;
      const notes = document.getElementById('notes').value;

      if (material || location || notes) {
        document.getElementById('additionalInfo').style.display = 'block';
        document.getElementById('resultMaterial').textContent = material ? 
          document.querySelector(`#materialType option[value="${material}"]`).textContent : 'Non renseigné';
        document.getElementById('resultLocation').textContent = location || 'Non renseigné';
        document.getElementById('resultNotes').textContent = notes || 'Aucune';
      }

      elements.resultSection.style.display = 'block';
      
      // Mettre à jour le statut
      elements.modelStatus.className = isCrackDetected ? 'model-status error' : 'model-status ready';
      elements.modelStatus.textContent = isCrackDetected ? '⚠️ Fissure détectée !' : '✅ Structure saine';

      // Scroll vers le résultat
      setTimeout(() => {
        elements.resultSection.scrollIntoView({ behavior: 'smooth', block: 'start' });
      }, 100);
    }

    // Analyse d'image
    async function analyzeImage(source = 'camera') {
      if (!model) {
        alert("Modèle IA non chargé");
        return;
      }

      let img = source === 'camera' ? elements.webcam : elements.image;
      
      canvas.width = img.videoWidth || img.naturalWidth;
      canvas.height = img.videoHeight || img.naturalHeight;
      ctx.drawImage(img, 0, 0, canvas.width, canvas.height);

      elements.modelStatus.className = 'model-status loading';
      elements.modelStatus.textContent = '🔄 Analyse en cours...';

      try {
        let predictions = await model.predict(canvas);
        let maxIndex = 0, maxProb = 0;
        
        predictions.forEach((pred, i) => {
          if (pred.probability > maxProb) {
            maxProb = pred.probability;
            maxIndex = i;
          }
        });

        displayResults(maxIndex, maxProb);
        
      } catch (error) {
        console.error('Erreur lors de l\'analyse:', error);
        elements.modelStatus.className = 'model-status error';
        elements.modelStatus.textContent = '❌ Erreur d\'analyse';
      }
    }

    // Gestion des fichiers uploadés
    function handleFile(file) {
      if (!file || !file.type.startsWith('image/')) {
        alert("Veuillez sélectionner une image valide");
        return;
      }
      
      const reader = new FileReader();
      reader.onload = (e) => {
        elements.image.src = e.target.result;
        elements.image.style.display = 'block';
        elements.webcam.style.display = 'none';
        elements.cameraInfo.style.display = 'none';
        elements.cameraSelection.style.display = 'none';
        analyzeImage('upload');
      };
      reader.readAsDataURL(file);
    }

    // Réinitialisation pour nouvelle analyse
    function resetAnalysis() {
      elements.resultSection.style.display = 'none';
      elements.image.style.display = 'none';
      elements.webcam.style.display = 'block';
      
      // Réafficher la sélection de caméra si nécessaire
      if (availableCameras.length > 1) {
        elements.cameraSelection.style.display = 'block';
      }
      
      // Réafficher les infos caméra si disponible
      if (stream && selectedCameraId) {
        elements.cameraInfo.style.display = 'block';
      }
      
      elements.modelStatus.className = 'model-status ready';
      elements.modelStatus.textContent = '✅ Modèle IA prêt !';
      
      document.getElementById('materialType').value = '';
      document.getElementById('location').value = '';
      document.getElementById('notes').value = '';
      document.getElementById('additionalInfo').style.display = 'none';
      
      window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    // Event Listeners
    elements.captureBtn.addEventListener('click', () => analyzeImage('camera'));
    elements.uploadBtn.addEventListener('click', () => elements.fileInput.click());
    elements.uploadArea.addEventListener('click', () => elements.fileInput.click());
    elements.fileInput.addEventListener('change', (e) => handleFile(e.target.files[0]));
    elements.newAnalysisBtn.addEventListener('click', resetAnalysis);

    // Switch camera
    elements.switchCameraBtn.addEventListener('click', async () => {
      currentFacingMode = currentFacingMode === "user" ? "environment" : "user";
      await setupWebcam();
    });

    // Drag & Drop
    elements.uploadArea.addEventListener('dragover', (e) => {
      e.preventDefault();
      elements.uploadArea.classList.add('dragover');
    });

    elements.uploadArea.addEventListener('dragleave', () => {
      elements.uploadArea.classList.remove('dragover');
    });

    elements.uploadArea.addEventListener('drop', (e) => {
      e.preventDefault();
      elements.uploadArea.classList.remove('dragover');
      handleFile(e.dataTransfer.files[0]);
    });

    // Nettoyage des ressources
    window.addEventListener('beforeunload', () => {
      if (stream) {
        stream.getTracks().forEach(track => track.stop());
      }
    });

    // Gestion responsive de l'orientation mobile
    if (isMobile) {
      window.addEventListener('orientationchange', () => {
        setTimeout(() => {
          if (stream) {
            setupWebcam();
          }
        }, 500);
      });
    }

    // Raccourcis clavier
    document.addEventListener('keydown', (e) => {
      if (e.code === 'Space' && e.target === document.body && stream && elements.webcam.style.display !== 'none') {
        e.preventDefault();
        elements.captureBtn.click();
      }
    });

    // Initialisation
    document.addEventListener('DOMContentLoaded', initializeModel);

    // Fallback si DOMContentLoaded déjà passé
    if (document.readyState === 'loading') {
      // Déjà géré par l'event listener ci-dessus
    } else {
      initializeModel();
    }
  </script>
</body>
</html>
